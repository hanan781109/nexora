async function buyPlan(planAmount) { if (!currentUser) { window.location.href = "login.html"; return; } planAmount = Number(planAmount); if (!Number.isFinite(planAmount) || planAmount <= 0) { showMessage("Invalid plan."); return; } const userRef = ref(db, `users/${currentUser.uid}`); const purchaseRef = push(ref(db, "planPurchases")); const now = Date.now(); try { const transactionResult = await runTransaction( userRef, (userData) => { if (userData === null) return; const wallet = Number(userData.wallet || 0); if (wallet < planAmount) return; const activePlansValue = Number(userData.activePlans); const activePlans = Number.isFinite(activePlansValue) ? activePlansValue : 0; const totalPlanPurchase = Number(userData.totalPlanPurchase || 0); return { ...userData, wallet: Number((wallet - planAmount).toFixed(8)), activePlans: activePlans + 1, totalPlanPurchase: Number( (totalPlanPurchase + planAmount).toFixed(8) ), lastPlanPurchase: planAmount, lastPlanPurchaseAt: now, updatedAt: now }; } ); if (!transactionResult.committed) { showMessage( `Insufficient balance. You need $${money(planAmount)} USDT.` ); return; } const buyerData = transactionResult.snapshot.val(); const purchaseData = { id: purchaseRef.key, uid: currentUser.uid, email: currentUser.email || "", planAmount: planAmount, currency: "USDT", status: "active", source: "wallet", createdAt: now, commissionStatus: "pending" }; await set(purchaseRef, purchaseData); const referrerUid = String(buyerData.referrerUid || "").trim(); const commissionRate = 0.20; const commission = Number( (planAmount * commissionRate).toFixed(8) ); if ( referrerUid && referrerUid !== currentUser.uid && commission > 0 ) { const referrerRef = ref(db, `users/${referrerUid}`); const commissionResult = await runTransaction( referrerRef, (referrerData) => { if (referrerData === null) return; const referrerWallet = Number( referrerData.wallet || 0 ); const referralEarnings = Number( referrerData.referralEarnings || 0 ); return { ...referrerData, wallet: Number( (referrerWallet + commission).toFixed(8) ), referralEarnings: Number( (referralEarnings + commission).toFixed(8) ), lastReferralCommission: commission, lastReferralCommissionAt: now, updatedAt: now }; } ); if (commissionResult.committed) { await set( ref(db, `referralCommissions/${purchaseRef.key}`), { id: purchaseRef.key, purchaseId: purchaseRef.key, referrerUid: referrerUid, referredUserUid: currentUser.uid, referredUserEmail: currentUser.email || "", planAmount: planAmount, rate: 20, commission: commission, currency: "USDT", createdAt: now } ); await update(purchaseRef, { commissionStatus: "paid", commission: commission, commissionRate: 20, referrerUid: referrerUid, commissionPaidAt: now }); showMessage( `$${money(planAmount)} plan purchased. $${money(commission)} referral commission added.` ); } else { await update(purchaseRef, { commissionStatus: "failed" }); showMessage( `$${money(planAmount)} plan purchased, but referral commission could not be added.` ); } } else { await update(purchaseRef, { commissionStatus: "not-applicable" }); showMessage( `$${money(planAmount)} plan purchased successfully.` ); } await loadWallet(currentUser.uid); } catch (error) { console.error("Plan purchase error:", error); showMessage( error.message || "Plan purchase failed. Please try again." ); } }
